cmake_minimum_required(VERSION 3.15)
project(gama_probe CXX)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_compile_definitions(_USE_MATH_DEFINES)
if(MSVC)
  add_compile_options(/EHsc /wd4244 /wd4267 /wd4477)
endif()
set(GAMA "${CMAKE_CURRENT_SOURCE_DIR}/../../../gama_tts" CACHE PATH "The engine source; point it at a patched copy to test the fixes")
add_subdirectory("${GAMA}" gama)
include_directories("${GAMA}/src" "${GAMA}/src/rapidxml" "${GAMA}/src/text_parser" "${GAMA}/src/vtm" "${GAMA}/src/vtm_control_model" "${GAMA}/src/xml")
add_executable(probe probe.cpp)
target_link_libraries(probe gamatts psapi)
